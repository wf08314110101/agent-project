fastapi              # API 框架
uvicorn[standard]    # ASGI 服务器
python-multipart     # FastAPI 文件上传

python-docx          # Word .docx 解析（备选，读段落）
openai               # OpenAI 兼容客户端 → 对接 DeepSeek
python-dotenv        # 加载 backend/.env
